Finding the right emoji is one of the most annoying micro-tasks on the web: open the OS picker, scroll forever, copy it. This page does the same job in one place — **every Unicode emoji, grouped by category**, with a live search and one-click copy.

<div id="ep-root" class="ep-root"></div>

## How to use

- **Search** in the box above. The list filters as you type, showing only the emojis that match.
- **Combine terms** — typing `happy dog` keeps only emojis whose name contains *both* words.
- **Jump to a category** by name too: try `flag`, `animal`, `food` or `smile`.
- **Click any emoji** to copy it to your clipboard. A small toast confirms the copy, and the tile pulses.
- **Clear or reset**: click the ✕, or press <kbd>Esc</kbd> while typing to wipe the search and show everything again.

## What's inside

The picker renders the full current Unicode set — nearly **2,000 emojis across the nine standard categories** (Smileys & Emotion, People & Body, Animals & Nature, Food & Drink, Travel & Places, Activities, Objects, Symbols, and Flags). The per-category count updates as you search, so you always know how many matches are left.

## A note on copying

Copying uses the browser's [Clipboard API](https://developer.mozilla.org/docs/Web/API/Clipboard_API) with a legacy fallback for older or insecure contexts. Pasted into most editors (chat, docs, code) the glyph lands as text, so it works in virtually everywhere — except a handful of rich-text fields that strip non-ASCII on paste. If a copy ever fails, the toast tells you and you can press <kbd>Ctrl/Cmd + C</kbd> on the highlighted tile.

// 3) Copy to clipboard, with a legacy fallback for older / insecure contexts.
async function copyText(text) {
  try {
    if (navigator.clipboard && navigator.clipboard.writeText) {
      await navigator.clipboard.writeText(text);
      return true;
    }
  } catch (e) {
    // fall through to the legacy path
  }
  try {
    const ta = document.createElement('textarea');
    ta.value = text;
    ta.setAttribute('readonly', '');
    ta.style.position = 'fixed';
    ta.style.left = '-9999px';
    document.body.appendChild(ta);
    ta.select();
    const ok = document.execCommand('copy');
    document.body.removeChild(ta);
    return ok;
  } catch (e) {
    return false;
  }
}

// 4) Build + wire one picker per mount point.
document.querySelectorAll('.ep-root').forEach((root) => {
  // Search bar
  const search = document.createElement('div');
  search.className = 'ep-search';
  const box = document.createElement('div');
  box.className = 'ep-search-box';
  const icon = document.createElement('span');
  icon.className = 'ep-search-icon';
  icon.setAttribute('aria-hidden', 'true');
  icon.textContent = '🔎';
  const input = document.createElement('input');
  input.type = 'search';
  input.className = 'ep-input';
  input.placeholder = 'Search emoji…  (try “heart”, “dog”, “flag”)';
  input.autocomplete = 'off';
  input.setAttribute('aria-label', 'Search emoji');
  const clearBtn = document.createElement('button');
  clearBtn.type = 'button';
  clearBtn.className = 'ep-clear';
  clearBtn.textContent = '✕';
  clearBtn.setAttribute('aria-label', 'Clear search');
  clearBtn.hidden = true;
  box.append(icon, input, clearBtn);
  const status = document.createElement('p');
  status.className = 'ep-status';
  status.setAttribute('role', 'status');
  status.setAttribute('aria-live', 'polite');
  search.append(box, status);

  // Grid + empty state
  const groupsEl = document.createElement('div');
  groupsEl.className = 'ep-groups';
  const empty = document.createElement('p');
  empty.className = 'ep-empty';
  empty.textContent = 'No emoji match your search.';
  empty.hidden = true;

  // Copy toast
  const toast = document.createElement('div');
  toast.className = 'ep-toast';
  toast.setAttribute('role', 'status');
  toast.setAttribute('aria-live', 'polite');
  const toastGlyph = document.createElement('span');
  toastGlyph.className = 'ep-toast-glyph';
  const toastText = document.createElement('span');
  toastText.className = 'ep-toast-text';
  toast.append(toastGlyph, toastText);
  let toastTimer = 0;

  // Build the category sections from the dataset.
  const frag = document.createDocumentFragment();
  const groupState = [];
  for (const g of emojiGroups) {
    const section = document.createElement('section');
    section.className = 'ep-group';
    const title = document.createElement('h3');
    title.className = 'ep-group-title';
    title.textContent = g.name;
    const count = document.createElement('span');
    count.className = 'ep-group-count';
    count.textContent = String(g.emojis.length);
    title.append(count);
    const grid = document.createElement('div');
    grid.className = 'ep-grid';
    const items = [];
    for (const pair of g.emojis) {
      const glyph = pair[0];
      const name = pair[1];
      const btn = document.createElement('button');
      btn.type = 'button';
      btn.className = 'ep-emoji';
      btn.dataset.emoji = glyph;
      btn.textContent = glyph;
      btn.title = name || glyph;
      btn.setAttribute('aria-label', 'Copy ' + (name || 'emoji'));
      grid.append(btn);
      // Also match on the category name, so "flag", "animal", "food" work.
      items.push({ el: btn, hay: (g.name + ' ' + (name || '')).toLowerCase() });
    }
    section.append(title, grid);
    frag.append(section);
    groupState.push({ section: section, count: count, items: items, full: g.emojis.length });
  }
  groupsEl.append(frag);
  root.append(search, groupsEl, empty, toast);

  // Live search: keep only the emojis whose haystack matches every token.
  function applyFilter() {
    const raw = input.value.trim().toLowerCase();
    const tokens = raw ? raw.split(/\s+/) : [];
    clearBtn.hidden = input.value.length === 0;
    let totalVisible = 0;
    for (const g of groupState) {
      let visible = 0;
      for (const it of g.items) {
        const show = tokens.every((t) => it.hay.includes(t));
        it.el.hidden = !show;
        if (show) visible = visible + 1;
      }
      g.section.hidden = visible === 0;
      g.count.textContent = String(tokens.length ? visible : g.full);
      totalVisible = totalVisible + visible;
    }
    empty.hidden = totalVisible > 0;
    if (!tokens.length) {
      status.textContent = totalEmojis.toLocaleString() + ' emoji in ' + emojiGroups.length + ' categories';
    } else if (totalVisible === 0) {
      status.textContent = 'No emoji match “' + input.value.trim() + '”';
    } else {
      status.textContent = totalVisible.toLocaleString() + ' result' + (totalVisible === 1 ? '' : 's') + ' for “' + input.value.trim() + '”';
    }
  }
  input.addEventListener('input', applyFilter);
  clearBtn.addEventListener('click', () => {
    input.value = '';
    applyFilter();
    input.focus();
  });
  input.addEventListener('keydown', (e) => {
    if (e.key === 'Escape') {
      input.value = '';
      applyFilter();
      input.blur();
    }
  });

  // Click any emoji to copy it (event-delegated for performance).
  groupsEl.addEventListener('click', async (e) => {
    const btn = e.target.closest('.ep-emoji');
    if (!btn || !btn.isConnected) return;
    const glyph = btn.dataset.emoji || '';
    const ok = await copyText(glyph);
    toastGlyph.textContent = glyph;
    toastText.textContent = ok ? 'Copied' : 'Copy failed — press Ctrl/Cmd+C';
    toast.classList.add('is-visible');
    clearTimeout(toastTimer);
    toastTimer = setTimeout(() => toast.classList.remove('is-visible'), 1500);
    if (ok) {
      btn.classList.remove('is-copied');
      void btn.offsetWidth; // restart the pulse animation
      btn.classList.add('is-copied');
      setTimeout(() => btn.classList.remove('is-copied'), 400);
    }
  });

  applyFilter();
});
</script>
